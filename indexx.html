// ================================================================
//  ESP32 — DÒ LINE BẰNG THUẬT TOÁN PID (BẢN THỰC CHIẾN)
//  TÍNH NĂNG: Live-Tuning qua Bluetooth / Serial Monitor
// ================================================================

#include "BluetoothSerial.h"

#if !defined(CONFIG_BT_ENABLED) || !defined(CONFIG_BLUEDROID_ENABLED)
#error Bluetooth is not enabled!
#endif
BluetoothSerial SerialBT;

#define UART_RX   16
#define UART_TX   17
#define ENC1_A    33
#define ENC1_B    32
#define ENC2_A    26
#define ENC2_B    25
#define PCT(p) (((int)((p) * 255 / 100)))

// --- THÔNG SỐ ĐIỀU CHỈNH QUA APP/SERIAL ---
int   BASE_SPEED = PCT(60);
float Kp = 4.3;  float Ki = 0.0;  float Kd = 3.5;

// --- THÔNG SỐ CỐ ĐỊNH ---
const int SPD_PIVOT      = PCT(85);
const unsigned long CONFIRM_MS = 100;
const unsigned long PIVOT_MS   = 400;
const int DIST_OBSTACLE  = 15;
const unsigned long NANO_TIMEOUT_MS = 500;

int lastError = 0;
int integral  = 0;
volatile long encCountL = 0;
volatile long encCountR = 0;

void IRAM_ATTR isrEncL() {
  if (digitalRead(ENC1_A) == digitalRead(ENC1_B)) encCountL--;
  else encCountL++;
}
void IRAM_ATTR isrEncR() {
  if (digitalRead(ENC2_A) == digitalRead(ENC2_B)) encCountR++;
  else encCountR--;
}

// ════ PARSE DỮ LIỆU TỪ NANO ════
void parseNanoData(const String& raw) {
  if (!raw.startsWith("#S")) return;
  int comma = raw.indexOf(',', 2);
  if (comma < 0) return;
  int d = raw.substring(2, comma).toInt();
  if (d > 0 && d < 400) distCm = d;
  String sStr = raw.substring(comma + 1);
  sStr.trim();
  if (sStr.length() < 5) return;
  for (int i = 0; i < 5; i++) {
    sensor[i] = (sStr[i] == '1') ? 1 : 0;
  }
  lastNanoMs = millis();
  nanoReady  = true;
}

// ════ PID LOGIC CHÍNH ════
void runLineFollow() {
  if (distCm > 0 && distCm < DIST_OBSTACLE) {
    sendMotorNow(0, 0); return;  // Dừng né vật cản
  }

  int code = getSensorCode();

  // Phát hiện góc vuông / ngã rẽ
  if (code == 0x1F) { sendMotorNow(0, 0); return; }     // 11111 → Dừng
  if (code == 0x1C || code == 0x1E || code == 0x18) { // Rẽ trái
    sendMotorNow(SPD_PIVOT, 0);
  }
  if (code == 0x07 || code == 0x0F || code == 0x03) { // Rẽ phải
    sendMotorNow(0, SPD_PIVOT);
  }

  // Tính error từ sensor
  int weights[5] = {-20, -10, 0, 10, 20};
  int sum = 0, active = 0;
  for (int i = 0; i < 5; i++) {
    if (sensor[i]) { sum += weights[i]; active++; }
  }
  int error = (active > 0) ? (sum / active) : (lastError > 0 ? 25 : -25);

  // PID Calculation
  int P = error;
  integral += error;
  integral = constrain(integral, -100, 100);
  int D = error - lastError;
  lastError = error;

  int offset = (Kp * P) + (Ki * integral) + (Kd * D);
  sendMotorNow(BASE_SPEED - offset, BASE_SPEED + offset);
}

void setup() {
  Serial.begin(115200);
  SerialBT.begin("Robot_Line_ESP32");
  Serial2.begin(38400, SERIAL_8N1, UART_RX, UART_TX);
  // ... setup encoder interrupts ...
}

void loop() {
  drainUART();
  checkBluetoothAndSerial();

  if (!nanoReady) return;
  runLineFollow();
  delay(5);
}

📡 Tại sao dùng 2 MCU (ESP32 + Nano)?

ESP32 lo PID, Bluetooth, xử lý logic phức tạp. Nano lo đọc cảm biến hồng ngoại + HC-SR04 siêu âm theo thời gian thực và gửi dữ liệu qua UART 38400 baud với định dạng #S<dist>,<sensors>. Phân tách nhiệm vụ giúp mỗi MCU chạy đúng sức mạnh của nó.

⚡ IRAM_ATTR cho ISR Encoder là gì?

Khi ESP32 truy cập Flash (đọc code, WiFi, BT...) thì Flash bị khóa tạm. Nếu ISR ở Flash → crash. IRAM_ATTR đặt ISR vào RAM nội (IRAM) → chạy ngay lập tức, không bao giờ bị block. Đặc biệt quan trọng cho encoder vì xung rất nhanh.

⚠️ Tại sao integral bị clamp ±100?

Nếu robot bị lạc đường lâu, integral sẽ tích lũy rất lớn gây hiện tượng integrator windup — khi quay lại đường thẳng, robot vẫn bẻ lái mạnh một thời gian dài. constrain(integral, -100, 100) giới hạn tích lũy, giúp phản ứng nhanh hơn khi tìm lại line.

🔥 sendMotorNow chỉ gửi khi thay đổi

So sánh currentLeft != lastSentL trước khi Serial2.printf giúp giảm lưu lượng UART đáng kể — đặc biệt khi robot đi thẳng (motor không đổi liên tục). Tránh saturate bộ đệm 38400 baud của Nano.

🔄 Tại sao Ki = 0 trong thực tế dò line?

Track dò line ngắn, sai số thường không tích lũy lâu. Ki tăng nguy cơ windup và làm robot dao động ở đường thẳng. Chỉ bật Ki khi có độ lệch hệ thống liên tục (ví dụ ma sát không đều 2 bánh) và cần fine-tune rất cẩn thận.

📲 Live-Tuning qua Bluetooth

Dùng app Serial Bluetooth Terminal (Android) hoặc bất kỳ BT terminal nào. Kết nối tên Robot_Line_ESP32, gửi lệnh P=5.5 + Enter. Thông số thay đổi ngay lập tức mà không cần nạp lại firmware — rất tiện khi chạy trên track thực.
